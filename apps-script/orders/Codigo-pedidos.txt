// The only browser-callable data method. Helpers ending in _ are private in Apps Script.
function chat(message, token, requestId) {
  if (typeof message !== 'string' || !message.trim() || message.length > 2000 ||
      typeof requestId !== 'string' || !/^[a-zA-Z0-9_-]{16,100}$/.test(requestId)) {
    return {ok:false, text:'Escribe una consulta de hasta 2000 caracteres.'};
  }
  var props = PropertiesService.getScriptProperties();
  var apiKey = props.getProperty('OPENAI_API_KEY');
  if (!apiKey) return {ok:false, text:'El chat está pendiente de activación.'};
  var cache = CacheService.getScriptCache();
  var lock = LockService.getScriptLock();
  var session, sid, secret, signedToken, reserved = false;
  try {
    if (!lock.tryLock(3000)) return {ok:false, text:'Estamos atendiendo otras consultas. Inténtalo en unos segundos.'};
    try {
      secret = props.getProperty('SESSION_SECRET');
      if (!secret) {
        secret = Utilities.getUuid() + Utilities.getUuid();
        props.setProperty('SESSION_SECRET',secret);
      }
      if (token) {
        var claims = verifyToken_(token,secret);
        sid = claims.sid;
        var saved = cache.get('chat:' + sid);
        if (!saved) throw new Error('SESSION_EXPIRED');
        session = JSON.parse(saved);
        if (session.expires !== claims.expires) throw new Error('SESSION_EXPIRED');
      } else {
        sid = Utilities.getUuid().replace(/-/g,'');
        session = {count:0, responseId:null, expires:Date.now() + SESSION_SECONDS_*1000};
      }
      signedToken = signToken_({sid:sid,expires:session.expires},secret);
      if (session.lastRequest === requestId && session.lastReply) {
        return {ok:true,text:session.lastReply,token:signedToken,remaining:MAX_TURNS_-session.count};
      }
      if (session.busy) return {ok:false,text:'La consulta anterior sigue en proceso. Espera su respuesta.',token:signedToken};
      if (session.count >= MAX_TURNS_) return {ok:false,text:'Esta conversación llegó a 20 consultas. Puedes iniciar un Nuevo chat.',token:signedToken};
      var day = Utilities.formatDate(new Date(),'America/Bogota','yyyy-MM-dd');
      var used = props.getProperty('USAGE_DAY') === day ? Number(props.getProperty('USAGE_COUNT') || 0) : 0;
      var cap = Number(props.getProperty('DAILY_REQUEST_LIMIT') || 300);
      if (!Number.isInteger(cap) || cap < 1 || cap > 1000) cap = 300;
      if (used >= cap) return {ok:false,text:'El chat alcanzó su límite diario de consultas. Vuelve mañana.'};
      // Count attempts before the API call, including failures. This caps public API usage.
      props.setProperties({USAGE_DAY:day,USAGE_COUNT:String(used+1)});
      session.busy = requestId;
      saveSession_(cache,sid,session);
      reserved = true;
    } finally { lock.releaseLock(); }

    var payload = requestPayload_(message.trim(),session.responseId);
    var response = UrlFetchApp.fetch('https://api.openai.com/v1/responses',{
      method:'post',contentType:'application/json',muteHttpExceptions:true,
      headers:{Authorization:'Bearer ' + apiKey,'OpenAI-Project':PROJECT_ID_},
      payload:JSON.stringify(payload)
    });
    var status = response.getResponseCode();
    var data;
    try { data = JSON.parse(response.getContentText()); }
    catch (e) { throw new Error('INVALID_API_RESPONSE'); }
    var result = parseResponse_(status,data);
    // Log only usage counts and error codes; no key, messages or product/order records.
    console.log(JSON.stringify({event:'openai_usage',input_tokens:data.usage && data.usage.input_tokens,
      output_tokens:data.usage && data.usage.output_tokens}));
    if (!lock.tryLock(3000)) throw new Error('SESSION_SAVE_FAILED');
    try {
      session.busy = null;
      session.count += 1;
      session.responseId = result.id;
      session.lastRequest = requestId;
      session.lastReply = result.text;
      saveSession_(cache,sid,session);
    } finally { lock.releaseLock(); }
    reserved = false;
    return {ok:true,text:result.text,token:signedToken,remaining:MAX_TURNS_-session.count};
  } catch (err) {
    console.error(JSON.stringify({event:'chat_error',code:err.code || err.message || 'UNKNOWN',status:err.status || null}));
    return {ok:false,text:customerError_(err),token:signedToken || null};
  } finally {
    if (reserved && sid) {
      // Release only our own reservation; never overwrite another request's state.
      if (lock.tryLock(3000)) {
        try {
          var current = cache.get('chat:' + sid);
          if (current) {
            current = JSON.parse(current);
            if (current.busy === requestId) { current.busy = null; saveSession_(cache,sid,current); }
          }
        } finally { lock.releaseLock(); }
      }
    }
  }
}

function parseResponse_(status,data) {
  if (status < 200 || status >= 300 || data.error || data.status !== 'completed') {
    var e = new Error('OPENAI_ERROR');
    e.status = status;
    e.code = (data.error && data.error.code) || (data.incomplete_details && data.incomplete_details.reason) || 'response_incomplete';
    throw e;
  }
  if (!/^resp_[a-zA-Z0-9_-]+$/.test(data.id || '')) throw new Error('INVALID_API_RESPONSE');
  var parts = [];
  (data.output || []).forEach(function(item) {
    if (item.type === 'mcp_approval_request') throw new Error('CATALOG_APPROVAL_REQUIRED');
    if (item.type === 'message') (item.content || []).forEach(function(part) {
      if (part.type === 'output_text' && typeof part.text === 'string') parts.push(part.text);
    });
  });
  if (!parts.length) throw new Error('EMPTY_RESPONSE');
  return {id:data.id,text:parts.join('\n\n')};
}

function customerError_(e) {
  if (e.code === 'rate_limit_exceeded' || e.code === 'slow_down') return 'Se alcanzó el límite temporal de OpenAI. Espera un momento y vuelve a consultar.';
  if (['insufficient_quota','credit_balance_exhausted','project_spend_limit_exceeded','organization_spend_limit_exceeded'].indexOf(e.code) >= 0) return 'El chat requiere una revisión de su saldo o presupuesto. Inténtalo más tarde.';
  if (e.message === 'SESSION_EXPIRED') return 'La conversación venció. Pulsa Nuevo chat.';
  if (e.status === 401 || e.status === 403) return 'La conexión con OpenAI necesita revisión. Inténtalo más tarde.';
  return 'No pude completar la consulta. Inténtalo nuevamente en un momento.';
}

function saveSession_(cache,sid,session) {
  var ttl = Math.floor((session.expires-Date.now())/1000);
  if (ttl <= 0) throw new Error('SESSION_EXPIRED');
  cache.put('chat:' + sid,JSON.stringify(session),Math.min(SESSION_SECONDS_,ttl));
}
function signToken_(claims,secret) {
  var payload = Utilities.base64EncodeWebSafe(JSON.stringify(claims),Utilities.Charset.UTF_8).replace(/=+$/,'');
  return payload + '.' + Utilities.base64EncodeWebSafe(Utilities.computeHmacSha256Signature(payload,secret)).replace(/=+$/,'');
}
function verifyToken_(token,secret) {
  try {
    if (typeof token !== 'string' || token.length > 1000) throw new Error();
    var parts = token.split('.');
    if (parts.length !== 2) throw new Error();
    var expected = Utilities.base64EncodeWebSafe(Utilities.computeHmacSha256Signature(parts[0],secret)).replace(/=+$/,'');
    var difference = expected.length ^ parts[1].length;
    for (var i=0;i<expected.length;i++) difference |= expected.charCodeAt(i) ^ (parts[1].charCodeAt(i) || 0);
    if (difference) throw new Error();
    var claims = JSON.parse(Utilities.newBlob(Utilities.base64DecodeWebSafe(parts[0])).getDataAsString());
    if (!/^[a-f0-9]{32}$/i.test(claims.sid || '') || !Number.isFinite(claims.expires) || claims.expires <= Date.now()) throw new Error();
    return claims;
  } catch (e) { throw new Error('SESSION_EXPIRED'); }
}

/** Two-field lookup: no personal data or workbook content is sent to OpenAI. */
var ORDER_NS_ = 'http://schemas.openxmlformats.org/spreadsheetml/2006/main';

function lookupOrder(type1,value1,type2,value2,product) {
  var fields=['name','phone','email','document'];
  if(fields.indexOf(type1)<0 || fields.indexOf(type2)<0 || type1===type2) return {ok:false,text:'Elige dos datos distintos de tu compra.'};
  var a=orderIdentity_(type1,value1),b=orderIdentity_(type2,value2);
  if(!a || !b) return {ok:false,text:'Revisa los dos datos. Escribe el nombre completo, el teléfono, el correo o la cédula tal como los registraste al comprar.'};
  if(product!==undefined && (typeof product!=='string' || product.length>200))return {ok:false,text:'Escribe un producto de hasta 200 caracteres.'};
  try {
    var props=PropertiesService.getScriptProperties(),lock=LockService.getScriptLock();
    if(!lock.tryLock(3000))return {ok:false,text:'Espera un momento y vuelve a consultar.'};
    try {
      var day=Utilities.formatDate(new Date(),'America/Bogota','yyyy-MM-dd');
      var used=props.getProperty('ORDER_DAY_V2')===day ? Number(props.getProperty('ORDER_ATTEMPTS_V2') || 0) : 0;
      if(used>=100)return {ok:false,text:'La consulta de guías alcanzó su límite diario. Inténtalo mañana o solicita atención en el canal donde compraste.'};
      props.setProperties({ORDER_DAY_V2:day,ORDER_ATTEMPTS_V2:String(used+1)});
    } finally {lock.releaseLock();}
    return matchingOrder_(readOrderWorkbook_(),type1,a,type2,b,product || '');
  } catch(e) {
    console.error(JSON.stringify({event:'order_lookup_error',code:'LOOKUP_UNAVAILABLE'}));
    return {ok:false,text:'No pude consultar la plantilla en este momento. Inténtalo más tarde o solicita atención en el canal donde compraste.'};
  }
}

function orderIdentity_(type,value) {
  if(typeof value!=='string' || value.length>254)return '';
  var s=value.trim();
  if(type==='email')return /^[^\s@,;<>]+@[^\s@,;<>]+\.[^\s@,;<>]+$/.test(s) ? s.toLowerCase() : '';
  if(type==='name') {
    s=orderHeader_(s).replace(/[.,]/g,'').replace(/\s+/g,' ');
    return s.length>=5 && s.length<=180 && s.split(' ').length>=2 ? s : '';
  }
  if(type==='phone' || type==='document') {
    if(!/^[+\d\s().-]+$/.test(s))return '';
    s=s.replace(/\D/g,'');
    if(type==='phone') {if(s.length===12 && s.slice(0,2)==='57')s=s.slice(2);return s.length===10 ? s : '';}
    return s.length>=5 && s.length<=15 ? s : '';
  }
  return '';
}

function matchingOrder_(book,type1,a,type2,b,product) {
  // Both fields must match the SAME row, not different records belonging to different people.
  var rows=book.rows.filter(function(r){return orderIdentity_(type1,String(r[type1] || ''))===a && orderIdentity_(type2,String(r[type2] || ''))===b;});
  var clue=orderHeader_(product);
  if(clue) {
    if(clue.length<4)return {ok:false,needsProduct:true,text:'Escribe al menos cuatro caracteres del producto comprado.'};
    rows=rows.filter(function(r){return orderHeader_(r.product).indexOf(clue)>=0;});
  }
  if(!rows.length)return {ok:false,text:'No encontré una compra que coincida con esos datos. Revísalos o prueba otra combinación. No compartas tus datos en el chat de productos.'};
  var groups={};rows.forEach(function(r){groups[r.platform+'|'+r.number]=true;});
  if(Object.keys(groups).length!==1)return {ok:false,needsProduct:true,text:'Esos datos coinciden con varias compras. Escribe el producto que estás esperando para identificar cuál buscas. Si sigue habiendo varias, solicita atención en el canal de compra.'};
  var seen={},guides=[];
  rows.forEach(function(r){if(r.guide && !seen[r.guide]){seen[r.guide]=true;guides.push(r.guide);}});
  // Guide only: never return contact info, name, document, address, products or other order IDs.
  return {ok:true,guides:guides,updated:book.updated,text:guides.length ? 'Guía registrada en la plantilla.' : 'Todavía no aparece una guía registrada para esta compra en la plantilla. Esto no confirma el estado del despacho.'};
}
function orderHeader_(value) {
  return String(value || '').normalize('NFD').replace(/[\u0300-\u036f]/g,'').trim().toUpperCase().replace(/\s+/g,' ');
}
function orderPlatform_(value) {
  var p=orderHeader_(value).replace(/[^A-Z0-9]/g,'');
  var map={JUMPSELLER:'WEB',PROVIPETS:'WEB',WEB:'WEB',DIRECTO:'DIRECTO',ADDI:'ADDI',PCO:'PCO',PUNTOSCOLOMBIA:'PCO',ML:'ML',MERCADOLIBRE:'ML',FALABELLA:'FALABELLA'};
  return map[p] || null;
}
function orderNumber_(value) {
  if (typeof value !== 'string' && typeof value !== 'number') return '';
  var s=String(value).trim().replace(/^#/,'').toUpperCase();
  return /^[A-Z0-9][A-Z0-9_-]{0,99}$/.test(s) ? s : '';
}
function readOrderWorkbook_() {
  var sourceId=PropertiesService.getScriptProperties().getProperty('ORDERS_FILE_ID');
  if(!sourceId)throw new Error('ORDERS_NOT_CONFIGURED');
  var file=DriveApp.getFileById(sourceId);
  var modified=file.getLastUpdated().getTime();
  if (file.getMimeType()!=='application/vnd.openxmlformats-officedocument.spreadsheetml.sheet') throw new Error('EXPECTED_XLSX');
  if (file.getSize()>10000000) throw new Error('FILE_TOO_LARGE');
  var cache=CacheService.getScriptCache();
  var prefix='orders-v2:'+modified+':';
  var meta=cache.get(prefix+'meta');
  if (meta) {
    meta=JSON.parse(meta);
    var chunks=[];
    for (var i=0;i<meta.count;i++) {var part=cache.get(prefix+i);if(part===null){chunks=null;break;}chunks.push(part);}
    if (chunks) return JSON.parse(chunks.join(''));
  }
  // Read XLSX directly as ZIP/XML: no conversion or edits to the PC-synced original.
  var blobs=Utilities.unzip(file.getBlob().setContentType('application/zip'));
  var entries={},total=0;
  blobs.forEach(function(b){var bytes=b.getBytes();total+=bytes.length;if(total>50000000)throw new Error('ARCHIVE_TOO_LARGE');entries[b.getName()]=b;});
  function xml(path){if(!entries[path])throw new Error('INVALID_XLSX');return XmlService.parse(entries[path].getDataAsString()).getRootElement();}
  var ns=XmlService.getNamespace(ORDER_NS_);
  var strings=[];
  if(entries['xl/sharedStrings.xml']) strings=xml('xl/sharedStrings.xml').getChildren('si',ns).map(function(si){return orderRichText_(si,ns);});
  var book={updated:new Date(modified).toISOString(),rows:[]};
  var names=Object.keys(entries).filter(function(n){return /^xl\/worksheets\/sheet\d+\.xml$/.test(n);});
  if(names.length>150)throw new Error('TOO_MANY_SHEETS');
  names.forEach(function(path){
    var data=xml(path).getChild('sheetData',ns);if(!data)return;
    var rows=data.getChildren('row',ns),headers=null;
    rows.forEach(function(row,index){
      var values={};
      row.getChildren('c',ns).forEach(function(cell){
        var attr=cell.getAttribute('r');if(!attr)return;
        var col=attr.getValue().replace(/[0-9]/g,'');
        var type=cell.getAttribute('t'),v=cell.getChildText('v',ns),is=cell.getChild('is',ns);
        values[col]=type && type.getValue()==='s' ? (strings[Number(v)] || '') : (is ? orderRichText_(is,ns) : (v || ''));
      });
      if(!headers && index<10){
        var candidate={};Object.keys(values).forEach(function(c){candidate[orderHeader_(values[c])]=c;});
        if(candidate.PLATAFORMA && (candidate.ORDEN || candidate.PEDIDO || candidate['NUMERO DE PEDIDO'])) headers=candidate;
        return;
      }
      if(!headers)return;
      function get(keys,max){for(var k=0;k<keys.length;k++){if(headers[keys[k]])return String(values[headers[keys[k]]] || '').trim().slice(0,max || 500);}return '';}
      var platform=orderPlatform_(get(['PLATAFORMA'],50));
      var number=orderNumber_(get(['ORDEN','PEDIDO','NUMERO DE PEDIDO'],100));
      if(!platform || !number)return;
      book.rows.push({platform:platform,number:number,email:get(['EMAIL','E-MAIL','CORREO','CORREO ELECTRONICO'],254).toLowerCase(),
        name:get(['NOMBRE','NOMBRE COMPLETO','CLIENTE'],180),phone:get(['TELEFONO','CELULAR','MOVIL'],40),document:get(['DOCUMENTO','CEDULA','IDENTIFICACION'],40),
        product:get(['PRODUCTO','NOMBRE PRODUCTO']),quantity:get(['UNDS','UNIDADES','CANTIDAD'],30),
        guide:get(['GUIA','NUMERO DE GUIA','NRO GUIA'],150),carrier:get(['TRANSPORTADORA','TRANSPORTADOR'],80)});
      if(book.rows.length>20000)throw new Error('TOO_MANY_ORDERS');
    });
  });
  if(!book.rows.length || file.getLastUpdated().getTime()!==modified)throw new Error('WORKBOOK_CHANGED');
  // Cache chunks below Google's per-key 100 KB limit, including UTF-8 characters.
  var serialized=JSON.stringify(book),count=Math.ceil(serialized.length/20000);
  if(count<=100){for(var c=0;c<count;c++)cache.put(prefix+c,serialized.slice(c*20000,(c+1)*20000),300);cache.put(prefix+'meta',JSON.stringify({count:count}),300);}
  return book;
}
function orderRichText_(node,ns) {
  var text=node.getChildText('t',ns) || '';
  node.getChildren('r',ns).forEach(function(r){text+=r.getChildText('t',ns) || '';});
  return text;
}
